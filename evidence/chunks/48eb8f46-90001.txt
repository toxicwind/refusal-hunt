 ────────────
  revdeps=$(pactree -r -d 1 "$owner" 2>/dev/null | tail -n +2 | tr '\n' ',' | sed 's/,$//')

  if [ -r /var/log/pacman.log ]; then

    last=$(grep -E "\[ALPM\] (installed|upgraded) $owner " /var/log/pacman.log 2>/dev/null | tail -1 | awk '{print $1, $2}')
    [ -n "$first" ] && log "pacman_log:$owner" "first_install" "$first"
    [ -n "$last" ]  && log "pacman_log:$owner" "last_change"   "$last"
  fi

  if command -v paru >/dev/null 2>&1; then
    paru -Qm "$owner" >/dev/null 2>&1 && log "aur:$owner" "source" "AUR (foreign)"
  fi

    r=$(expac '%r' "$owner" 2>/dev/null)

  fi
